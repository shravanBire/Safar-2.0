class PreferencesDataStore(private val context: Context) {

    companion object {
        val KEY_IS_DARK_THEME = booleanPreferencesKey("is_dark_theme")
        val KEY_SPEED_UNIT = stringPreferencesKey("speed_unit")

        const val UNIT_KMH = "km/h"
        const val UNIT_MPH = "mph"
    }

    val isDarkTheme: Flow<Boolean> = context.dataStore.data
        .map { preferences ->
            preferences[KEY_IS_DARK_THEME] ?: true // Default to Midnight Moto Cockpit Dark Theme
        }

    val speedUnit: Flow<String> = context.dataStore.data
        .map { preferences ->
            preferences[KEY_SPEED_UNIT] ?: UNIT_KMH
        }

    suspend fun setDarkTheme(isDark: Boolean) {
        context.dataStore.edit { preferences ->
            preferences[KEY_IS_DARK_THEME] = isDark
        }
    }

    suspend fun setSpeedUnit(unit: String) {
        context.dataStore.edit { preferences ->
            preferences[KEY_SPEED_UNIT] = if (unit == UNIT_MPH) UNIT_MPH else UNIT_KMH
        }
    }

    suspend fun toggleSpeedUnit() {
        context.dataStore.edit { preferences ->
            val current = preferences[KEY_SPEED_UNIT] ?: UNIT_KMH
            preferences[KEY_SPEED_UNIT] = if (current == UNIT_KMH) UNIT_MPH else UNIT_KMH
        }
    }
}

package com.example.safar20.data.local

import java.util.Locale

object UnitConverter {

    const val KM_TO_MILES_FACTOR: Double = 0.621371
    const val UNIT_KMH = "km/h"
    const val UNIT_MPH = "mph"

    /**
     * Converts speed from km/h to selected unit (mph if unit is "mph").
     */
    fun convertSpeed(speedKmh: Double, unit: String): Double {
        return if (isMph(unit)) {
            speedKmh * KM_TO_MILES_FACTOR
        } else {
            speedKmh
        }
    }

    /**
     * Converts Float speed from km/h to selected unit (mph if unit is "mph").
     */
    fun convertSpeed(speedKmh: Float, unit: String): Float {
        return if (isMph(unit)) {
            (speedKmh * KM_TO_MILES_FACTOR).toFloat()
        } else {
            speedKmh
        }
    }

    /**
     * Converts distance from km to selected unit (miles if unit is "mph").
     */
    fun convertDistance(distanceKm: Double, unit: String): Double {
        return if (isMph(unit)) {
            distanceKm * KM_TO_MILES_FACTOR
        } else {
            distanceKm
        }
    }

    /**
     * Formats speed value with unit label (e.g. "65 km/h" or "40 mph").
     */
    fun formatSpeed(speedKmh: Double, unit: String, decimals: Int = 0): String {
        val converted = convertSpeed(speedKmh, unit)
        val label = getSpeedUnitLabel(unit)
        return String.format(Locale.US, "%.${decimals}f %s", converted, label)
    }

    /**
     * Formats distance value with unit label (e.g. "12.4 km" or "7.7 mi").
     */
    fun formatDistance(distanceKm: Double, unit: String, decimals: Int = 1): String {
        val converted = convertDistance(distanceKm, unit)
        val label = getDistanceUnitLabel(unit)
        return String.format(Locale.US, "%.${decimals}f %s", converted, label)
    }

    /**
     * Returns speed unit display label ("km/h" or "mph").
     */
    fun getSpeedUnitLabel(unit: String): String {
        return if (isMph(unit)) "mph" else "km/h"
    }

    /**
     * Returns distance unit display label ("km" or "mi").
     */
    fun getDistanceUnitLabel(unit: String): String {
        return if (isMph(unit)) "mi" else "km"
    }

    private fun isMph(unit: String): Boolean {
        return unit.equals(UNIT_MPH, ignoreCase = true)
    }
}

package com.example.safar20.data.network

import java.nio.ByteBuffer
import java.nio.ByteOrder

/**
 * Binary Packet Codec for SAFAR 2.0 Wire Protocol.
 * Handles encoding/decoding of 16-byte packed TripPoint structs,
 * 20-byte live telemetry packets, 488-byte history chunks, and CRC16 checksums.
 */
object BinaryPacketCodec {

    const val OPCODE_OPEN_SESSION: Byte = 0x01
    const val OPCODE_LIVE_TELEMETRY: Byte = 0x02
    const val OPCODE_HISTORY_CHUNK: Byte = 0x03
    const val OPCODE_HISTORY_ACK: Byte = 0x04
    const val OPCODE_HEARTBEAT: Byte = 0x05

    const val TRIP_POINT_SIZE = 16
    const val LIVE_TELEMETRY_SIZE = 21 // 1B Opcode + 2B Seq + 16B TripPoint + 2B CRC16
    const val HISTORY_CHUNK_SIZE = 488  // 1B Opcode + 2B ChunkID + 2B TotalChunks + 1B Count + 480B Points + 2B CRC16

    data class ParsedTripPoint(
        val timestamp: Long,        // Unix Epoch Seconds (UTC)
        val latitude: Double,       // Latitude in degrees
        val longitude: Double,      // Longitude in degrees
        val speedKmh: Float,        // Speed in km/h
        val heading: Int,           // Course angle 0..359 degrees
        val isIgnitionOn: Boolean,  // Bit 0 of flags
        val isTripStart: Boolean,   // Bit 1 of flags
        val isTripEnd: Boolean,     // Bit 2 of flags
        val satCount: Int           // Bits 3..7 of flags
    )

    data class LiveTelemetryPacket(
        val sequenceNumber: Int,
        val tripPoint: ParsedTripPoint,
        val isValidCrc: Boolean
    )

    data class HistoryChunkPacket(
        val chunkId: Int,
        val totalChunks: Int,
        val count: Int,
        val tripPoints: List<ParsedTripPoint>,
        val isValidCrc: Boolean
    )

    /**
     * Decodes a 16-byte packed TripPoint struct from raw bytes at offset.
     * Uses Little Endian byte order matching ESP32 GCC struct layout.
     */
    fun decodeTripPoint(data: ByteArray, offset: Int = 0): ParsedTripPoint {
        val buffer = ByteBuffer.wrap(data, offset, TRIP_POINT_SIZE).order(ByteOrder.LITTLE_ENDIAN)

        // uint32_t timestamp (Bytes 0..3)
        val timestamp = buffer.int.toLong() and 0xFFFFFFFFL

        // int32_t lat_e7 (Bytes 4..7)
        val latE7 = buffer.int
        val latitude = latE7 / 10_000_000.0

        // int32_t lon_e7 (Bytes 8..11)
        val lonE7 = buffer.int
        val longitude = lonE7 / 10_000_000.0

        // uint16_t speed_decideg (Bytes 12..13) -> Speed in km/h * 10
        val speedDecideg = buffer.short.toInt() and 0xFFFF
        val speedKmh = speedDecideg / 10.0f

        // uint8_t heading_div2 (Byte 14) -> Heading / 2
        val headingDiv2 = buffer.get().toInt() and 0xFF
        val heading = (headingDiv2 * 2) % 360

        // uint8_t flags (Byte 15)
        val flags = buffer.get().toInt() and 0xFF
        val isIgnitionOn = (flags and 0x01) != 0
        val isTripStart = (flags and 0x02) != 0
        val isTripEnd = (flags and 0x04) != 0
        val satCount = (flags shr 3) and 0x1F

        return ParsedTripPoint(
            timestamp = timestamp,
            latitude = latitude,
            longitude = longitude,
            speedKmh = speedKmh,
            heading = heading,
            isIgnitionOn = isIgnitionOn,
            isTripStart = isTripStart,
            isTripEnd = isTripEnd,
            satCount = satCount
        )
    }

    /**
     * Encodes a ParsedTripPoint into a 16-byte packed byte array (Little Endian).
     */
    fun encodeTripPoint(point: ParsedTripPoint): ByteArray {
        val buffer = ByteBuffer.allocate(TRIP_POINT_SIZE).order(ByteOrder.LITTLE_ENDIAN)

        buffer.putInt((point.timestamp and 0xFFFFFFFFL).toInt())
        buffer.putInt((point.latitude * 10_000_000.0).toInt())
        buffer.putInt((point.longitude * 10_000_000.0).toInt())
        buffer.putShort(((point.speedKmh * 10.0f).toInt() and 0xFFFF).toShort())
        buffer.put((point.heading / 2).toByte())

        var flags = 0
        if (point.isIgnitionOn) flags = flags or 0x01
        if (point.isTripStart) flags = flags or 0x02
        if (point.isTripEnd) flags = flags or 0x04
        flags = flags or ((point.satCount and 0x1F) shl 3)

        buffer.put(flags.toByte())

        return buffer.array()
    }

    /**
     * Decodes 20/21-byte Live Telemetry Packet (0x02).
     */
    fun decodeLiveTelemetry(data: ByteArray): LiveTelemetryPacket? {
        if (data.size < 19) return null

        val offset = if (data[0] == OPCODE_LIVE_TELEMETRY) 1 else 0
        if (data.size < offset + 18) return null

        val buffer = ByteBuffer.wrap(data, offset, 2).order(ByteOrder.LITTLE_ENDIAN)
        val seqNum = buffer.short.toInt() and 0xFFFF

        val tripPoint = decodeTripPoint(data, offset + 2)

        val isValidCrc = if (data.size >= offset + 20) {
            val crcCalculated = computeCrc16(data, 0, data.size - 2)
            val crcReceived = ((data[data.size - 2].toInt() and 0xFF) or ((data[data.size - 1].toInt() and 0xFF) shl 8))
            crcCalculated == crcReceived
        } else true

        return LiveTelemetryPacket(
            sequenceNumber = seqNum,
            tripPoint = tripPoint,
            isValidCrc = isValidCrc
        )
    }

    /**
     * Encodes Live Telemetry Packet (0x02).
     */
    fun encodeLiveTelemetry(sequenceNumber: Int, point: ParsedTripPoint): ByteArray {
        val payload = ByteBuffer.allocate(LIVE_TELEMETRY_SIZE).order(ByteOrder.LITTLE_ENDIAN)
        payload.put(OPCODE_LIVE_TELEMETRY)
        payload.putShort((sequenceNumber and 0xFFFF).toShort())

        val pointBytes = encodeTripPoint(point)
        payload.put(pointBytes)

        val crc = computeCrc16(payload.array(), 0, LIVE_TELEMETRY_SIZE - 2)
        payload.putShort(crc.toShort())

        return payload.array()
    }

    /**
     * Decodes 488-byte History Chunk Packet (0x03).
     */
    fun decodeHistoryChunk(data: ByteArray): HistoryChunkPacket? {
        if (data.size < 6) return null

        val offset = if (data[0] == OPCODE_HISTORY_CHUNK) 1 else 0
        val buffer = ByteBuffer.wrap(data, offset, 5).order(ByteOrder.LITTLE_ENDIAN)

        val chunkId = buffer.short.toInt() and 0xFFFF
        val totalChunks = buffer.short.toInt() and 0xFFFF
        val count = buffer.get().toInt() and 0xFF

        val pointsList = mutableListOf<ParsedTripPoint>()
        var pointOffset = offset + 5

        for (i in 0 until minOf(count, 30)) {
            if (pointOffset + TRIP_POINT_SIZE <= data.size) {
                pointsList.add(decodeTripPoint(data, pointOffset))
                pointOffset += TRIP_POINT_SIZE
            }
        }

        val isValidCrc = if (data.size >= HISTORY_CHUNK_SIZE) {
            val crcCalculated = computeCrc16(data, 0, data.size - 2)
            val crcReceived = ((data[data.size - 2].toInt() and 0xFF) or ((data[data.size - 1].toInt() and 0xFF) shl 8))
            crcCalculated == crcReceived
        } else true

        return HistoryChunkPacket(
            chunkId = chunkId,
            totalChunks = totalChunks,
            count = count,
            tripPoints = pointsList,
            isValidCrc = isValidCrc
        )
    }

    /**
     * Encodes 488-byte History Chunk Packet (0x03).
     */
    fun encodeHistoryChunk(chunkId: Int, totalChunks: Int, points: List<ParsedTripPoint>): ByteArray {
        val buffer = ByteBuffer.allocate(HISTORY_CHUNK_SIZE).order(ByteOrder.LITTLE_ENDIAN)
        buffer.put(OPCODE_HISTORY_CHUNK)
        buffer.putShort((chunkId and 0xFFFF).toShort())
        buffer.putShort((totalChunks and 0xFFFF).toShort())

        val count = minOf(points.size, 30)
        buffer.put(count.toByte())

        for (i in 0 until count) {
            buffer.put(encodeTripPoint(points[i]))
        }

        val remainingBytes = 480 - (count * TRIP_POINT_SIZE)
        if (remainingBytes > 0) {
            buffer.put(ByteArray(remainingBytes))
        }

        val crc = computeCrc16(buffer.array(), 0, HISTORY_CHUNK_SIZE - 2)
        buffer.putShort(crc.toShort())

        return buffer.array()
    }

    /**
     * Calculates CRC16-CCITT (0x1021) polynomial checksum.
     */
    fun computeCrc16(data: ByteArray, offset: Int = 0, length: Int = data.size): Int {
        var crc = 0xFFFF
        for (i in offset until (offset + length)) {
            crc = crc xor ((data[i].toInt() and 0xFF) shl 8)
            for (j in 0 until 8) {
                crc = if ((crc and 0x8000) != 0) {
                    (crc shl 1) xor 0x1021
                } else {
                    crc shl 1
                }
            }
        }
        return crc and 0xFFFF
    }

    /**
     * Verifies if last 2 bytes of payload match calculated CRC16.
     */
    fun verifyCrc16(data: ByteArray): Boolean {
        if (data.size < 2) return false
        val calculated = computeCrc16(data, 0, data.size - 2)
        val received = ((data[data.size - 2].toInt() and 0xFF) or ((data[data.size - 1].toInt() and 0xFF) shl 8))
        return calculated == received
    }
}

private fun DrawScope.drawDialTicksAndLabels(
    center: Offset,
    radius: Float,
    startAngle: Float,
    totalSweep: Float
) {
    for (i in 0..160 step 10) {
        val speedRatio = i / 160f
        val tickAngle = startAngle + (speedRatio * totalSweep)
        val tickRad = Math.toRadians(tickAngle.toDouble())

        val isMajor = (i % 20 == 0)
        val isRedline = (i >= 120)

        val tickLength = if (isMajor) 12.dp.toPx() else 6.dp.toPx()
        val tickWidth = if (isMajor) 2.5.dp.toPx() else 1.2.dp.toPx()
        val tickColor = if (isRedline) SignalRed else if (isMajor) RoyalBrass else BronzeRim

        val innerPt = Offset(
            x = center.x + ((radius - tickLength) * cos(tickRad)).toFloat(),
            y = center.y + ((radius - tickLength) * sin(tickRad)).toFloat()
        )
        val outerPt = Offset(
            x = center.x + (radius * cos(tickRad)).toFloat(),
            y = center.y + (radius * sin(tickRad)).toFloat()
        )

        drawLine(
            color = tickColor,
            start = innerPt,
            end = outerPt,
            strokeWidth = tickWidth,
            cap = StrokeCap.Round
        )

import androidx.compose.foundation.layout.fillMaxSize

import com.example.safar20.data.local.entity.LocationPointEntity
import org.maplibre.android.MapLibre
import org.maplibre.android.camera.CameraUpdateFactory
import org.maplibre.android.geometry.LatLng
import org.maplibre.android.geometry.LatLngBounds

/**
 * MapLibre Native MapView for Cockpit & Trips preview.
 * Polyline and marker updates occur strictly inside the AndroidView `update` block per SAFAR 2.0 specs.
 */
@Composable
fun CockpitMapView(
    latitude: Double,
    longitude: Double,
    heading: Int = 0,
    routePoints: List<LocationPointEntity> = emptyList(),
    modifier: Modifier = Modifier.fillMaxSize()
) {
    val context = LocalContext.current
    remember { MapLibre.getInstance(context) }

    val mapView = remember { MapView(context) }
    val lifecycleOwner = LocalLifecycleOwner.current

    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            when (event) {
                Lifecycle.Event.ON_START -> mapView.onStart()
                Lifecycle.Event.ON_RESUME -> mapView.onResume()
                Lifecycle.Event.ON_PAUSE -> mapView.onPause()
                Lifecycle.Event.ON_STOP -> mapView.onStop()
                Lifecycle.Event.ON_DESTROY -> mapView.onDestroy()
                else -> {}
            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose {
            lifecycleOwner.lifecycle.removeObserver(observer)
        }
    }

    AndroidView(
        factory = {
            mapView.apply {
                getMapAsync { map ->
                    map.setStyle(Style.Builder().fromUri("https://demotiles.maplibre.org/style.json")) {
                        map.uiSettings.isCompassEnabled = false
                        map.uiSettings.isAttributionEnabled = false
                        map.uiSettings.isLogoEnabled = false

                        val initialPos = LatLng(latitude, longitude)
                        map.animateCamera(CameraUpdateFactory.newLatLngZoom(initialPos, 14.0))
                    }
                }
            }
        },
        update = { view ->
            view.getMapAsync { map ->
                map.getStyle {
                    if (routePoints.isNotEmpty()) {
                        val latLngList = routePoints.map { LatLng(it.latitude, it.longitude) }

                        if (latLngList.size >= 2) {
                            try {
                                val boundsBuilder = LatLngBounds.Builder()
                                latLngList.forEach { boundsBuilder.include(it) }
                                map.animateCamera(CameraUpdateFactory.newLatLngBounds(boundsBuilder.build(), 100))
                            } catch (e: Exception) {
                                // Fallback
                                map.animateCamera(CameraUpdateFactory.newLatLngZoom(latLngList.first(), 14.0))
                            }
                        }
                    } else if (latitude != 0.0 && longitude != 0.0) {
                        val currentPos = LatLng(latitude, longitude)
                        map.animateCamera(CameraUpdateFactory.newLatLngZoom(currentPos, 15.0))
                    }
                }
            }
        },
        modifier = modifier
    )
}

/**
 * Royal Enfield Retro-Moto Cockpit Theme for SAFAR 2.0.
 * Defaults to Midnight Moto Cockpit dark theme.
 */
@Composable
fun SafarTheme(
    darkTheme: Boolean = true,
    content: @Composable () -> Unit
) {
    val colorScheme = if (darkTheme) DarkColorScheme else LightColorScheme
    val view = LocalView.current

    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = colorScheme.background.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !darkTheme
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}

/**
 * Compatibility alias for Safar20Theme.
 */
@Composable
fun Safar20Theme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit
) {
    SafarTheme(darkTheme = darkTheme, content = content)
}

package com.example.safar20.viewModels

import android.content.Context
import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.lifecycle.viewModelScope
import com.example.safar20.data.local.entity.DeviceEntity
import com.example.safar20.data.local.entity.LocationPointEntity
import com.example.safar20.data.local.entity.TripSummaryEntity
import com.example.safar20.data.network.BinaryPacketCodec
import com.example.safar20.data.repository.SafarRepository
import com.example.safar20.ui.components.ConnectionState
import kotlinx.coroutines.ExperimentalCoroutinesApi
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.flatMapLatest
import kotlinx.coroutines.flow.flowOf
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

/**
 * Shared ViewModel for SAFAR 2.0.
 * Serves HomeScreen, TripsScreen, and SettingsScreen with a single source of truth.
 */
@OptIn(ExperimentalCoroutinesApi::class)
class SafarViewModel(
    private val repository: SafarRepository
) : ViewModel() {

    // --- Core StateFlows from Repository ---
    val allDevices: StateFlow<List<DeviceEntity>> = repository.allDevices
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    val selectedDevice: StateFlow<DeviceEntity?> = repository.selectedDevice
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), null)

    val isDarkTheme: StateFlow<Boolean> = repository.isDarkTheme
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), true)

    val speedUnit: StateFlow<String> = repository.speedUnit
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), "km/h")

    // --- Live Telemetry State ---
    private val _liveTelemetry = MutableStateFlow<BinaryPacketCodec.ParsedTripPoint?>(null)
    val liveTelemetry: StateFlow<BinaryPacketCodec.ParsedTripPoint?> = _liveTelemetry.asStateFlow()

    private val _isSimulatorRunning = MutableStateFlow(false)
    val isSimulatorRunning: StateFlow<Boolean> = _isSimulatorRunning.asStateFlow()

    val connectionState: StateFlow<ConnectionState> = combine(
        selectedDevice,
        _isSimulatorRunning,
        _liveTelemetry
    ) { device, isSim, telemetry ->
        when {
            device == null -> ConnectionState.OFFLINE
            isSim -> ConnectionState.SIMULATOR_SYNCING
            telemetry != null && (System.currentTimeMillis() / 1000 - telemetry.timestamp) < 10 -> ConnectionState.P2P_DIRECT
            else -> ConnectionState.PARKED
        }
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), ConnectionState.PARKED)

    // --- Trip History State ---
    val tripHistory: StateFlow<List<TripSummaryEntity>> = selectedDevice
        .flatMapLatest { device ->
            if (device != null) {
                repository.getTripsForDevice(device.deviceId)
            } else {
                flowOf(emptyList())
            }
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    private val _selectedTripId = MutableStateFlow<String?>(null)
    val selectedTripId: StateFlow<String?> = _selectedTripId.asStateFlow()

    val selectedTripPoints: StateFlow<List<LocationPointEntity>> = _selectedTripId
        .flatMapLatest { tripId ->
            if (tripId != null) {
                repository.getPointsForTrip(tripId)
            } else {
                flowOf(emptyList())
            }
        }
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    init {
        // Collect live telemetry from Repository
        viewModelScope.launch {
            repository.liveTelemetryStream.collect { telemetryPoint ->
                _liveTelemetry.value = telemetryPoint
                selectedDevice.value?.let { device ->
                    repository.updateLastKnownLocation(
                        deviceId = device.deviceId,
                        lat = telemetryPoint.latitude,
                        lon = telemetryPoint.longitude,
                        speed = telemetryPoint.speedKmh,
                        timestamp = telemetryPoint.timestamp
                    )
                }
            }
        }

        // Auto-seed mock data on initial launch
        viewModelScope.launch {
            repository.seedMockData("SAFAR_01")
        }
    }

    // --- UI Actions ---
    fun selectDevice(deviceId: String) {
        viewModelScope.launch {
            repository.setSelectedDevice(deviceId)
            _selectedTripId.value = null
        }
    }

    fun toggleDarkTheme(isDark: Boolean) {
        viewModelScope.launch {
            repository.setDarkTheme(isDark)
        }
    }

    fun setSpeedUnit(unit: String) {
        viewModelScope.launch {
            repository.setSpeedUnit(unit)
        }
    }

    fun toggleSpeedUnit() {
        viewModelScope.launch {
            repository.toggleSpeedUnit()
        }
    }

    fun toggleSimulator(enable: Boolean) {
        _isSimulatorRunning.value = enable
        if (enable) {
            repository.startSimulator()
        } else {
            repository.stopSimulator()
        }
    }

    fun selectTripForMapPreview(tripId: String?) {
        _selectedTripId.value = tripId
    }

    fun pairDeviceFromQr(
        qrUriString: String,
        onSuccess: (DeviceEntity) -> Unit,
        onError: (String) -> Unit
    ) {
        viewModelScope.launch {
            val result = repository.parseAndPairQrUri(qrUriString)
            result.onSuccess { device ->
                onSuccess(device)
            }.onFailure { exception ->
                onError(exception.message ?: "Failed to parse QR URI")
            }
        }
    }

    fun addManualDevice(
        id: String,
        name: String,
        simNumber: String,
        onSuccess: (DeviceEntity) -> Unit
    ) {
        viewModelScope.launch {
            val device = repository.addManualDevice(id, name, simNumber)
            onSuccess(device)
        }
    }

    fun deleteDevice(deviceId: String) {
        viewModelScope.launch {
            repository.deleteDevice(deviceId)
        }
    }

    class Factory(private val context: Context) : ViewModelProvider.Factory {
        @Suppress("UNCHECKED_CAST")
        override fun <T : ViewModel> create(modelClass: Class<T>): T {
            val repo = SafarRepository.getInstance(context)
            return SafarViewModel(repo) as T
        }
    }
